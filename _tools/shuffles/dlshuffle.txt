  function dlShuffle(then) {   // a different shuffle each time (shared engine, see _tools/shuffles); the old rotating deck is the fallback
    dlSub = null; var tok = ++dlTok;
    if (typeof apShuffle === 'undefined') { var d = ''; for (var k = 0; k < 7; k++) d += '<div class="dl-dk" style="--a:' + ((k - 3) * 11) + 'deg;--s:' + (k % 2 ? 30 : -30) + '%"></div>'; frame('Deal it', '<div class="dl-deck">' + d + '</div><div class="dl-shuf">Shuffling…</div>'); dplay('fan1'); dTm(function () { dplay('fan2'); }, 800); dTm(then, 1700); return; }
    frame('Deal it', '<div class="shx-st" id="shx-st" style="width:min(92%,880px)"></div><div class="dl-shuf">Shuffling…</div>');
    var st = $('shx-st'), fired = false;
    apShuffle.run(st, { ar: 2 / 3, back: 'var(--cb) center/cover no-repeat,#0d1a38', faces: [], later: dTm, play: dplay,
      alive: function () { return !!(ov && st && st.isConnected && tok === dlTok); },
      done: function () { if (fired) return; fired = true; dTm(then, 0); } });
  }
